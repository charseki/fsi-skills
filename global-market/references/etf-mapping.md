## Python 查询函数（用于收盘日报）
```python
def query_fund_chg(code):
    """通过天天基金API查询估算涨跌幅"""
    import requests, re, json
    try:
        r = requests.get(f"https://fundgz.1234567.com.cn/js/{code}.js", timeout=10)
        match = re.search(r'jsonpgz\((.+?)\)\s*;?\s*$', r.text)
        if match:
            data = json.loads(match.group(1))
            return float(data.get("gszzl", 0) or 0)
    except:
        pass
    return None
```
