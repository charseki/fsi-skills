def query_fund_current(code):
    """Get current estimated NAV from fundgz API"""
    try:
        r = requests.get(f"https://fundgz.1234567.com.cn/js/{code}.js", timeout=10)
        if r.status_code == 200:
            match = re.search(r'jsonpgz\((.+?)\)\s*;?\s*$', r.text)
            if match:
                data = json.loads(match.group(1))
                return float(data.get("gszzl", 0) or 0)
    except:
        pass
    return None


def query_fund_current_full(code):
    """Get full current NAV info"""
    try:
        r = requests.get(f"https://fundgz.1234567.com.cn/js/{code}.js", timeout=10)
        if r.status_code == 200:
            match = re.search(r'jsonpgz\((.+?)\)\s*;?\s*$', r.text)
            if match:
                return json.loads(match.group(1))
    except:
        pass
    return None


def query_fund_history(code):
    """Get historical NAV for a fund (returns DataFrame)"""
    try:
        df = ak.fund_open_fund_info_em(symbol=code, indicator="单位净值走势")
        if df is not None and len(df) > 0:
            df["净值日期"] = pd.to_datetime(df["净值日期"])
            return df
    except:
        pass
    return None


def get_sector_fund_chg(sector):
    """Get current daily change for a sector's fund"""
    if sector not in FUND_MAP:
        return None, None
    code, name = FUND_MAP[sector]
    chg = query_fund_current(code)
    return chg, (code, name)
